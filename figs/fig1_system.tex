% ===================================================================
%  Figure 1 -- ActDelta system overview.
%
%  Drawn at true size: the picture is 84 mm wide, which is \columnwidth for
%  acmart sigconf at 10 pt, so it is never rescaled and no label falls
%  below 7 pt.
%
%  The three facts the figure has to carry, in order of importance:
%    1. only the agent view is scheduled; wrist view and proprioception
%       stay real on every step (this is what bounds the claim);
%    2. u_t is decided from the robot-side latent state alone -- the policy
%       appears only on the edge side, so there is no local policy call;
%    3. the executed action feeds the world model forward, so a suppression
%       decision changes the future observation distribution.
%
%  Requires figs/figstyles.tex.
% ===================================================================
\begin{tikzpicture}[x=1mm,y=1mm]

% ---------------------------------------------------------------- bands
\fill[fgBlue!5, band] (0,31) rectangle (84,51.5);
\fill[black!4,  band] (0,6.3) rectangle (84,19.5);

\node[tag, text=fgBlue, anchor=west] at (1.2,49.0) {Robot};
\node[lblb, anchor=west]             at (11.0,49.0) {agent view $o_t$ scheduled};
\node[tag, text=fgGrey, anchor=east] at (80,18.0) {Edge server};

% ------------------------------------------------------- robot-side chain
\camglyph{(5,41.25)}{fgBlue}

\node[bxb] (enc)  at (19.25,41.25) [minimum width=15.5mm, minimum height=7.5mm]
     {Encoder\\$z_t$};
\node[bxb] (wm)   at (37.25,41.25) [minimum width=15.5mm, minimum height=7.5mm]
     {World model\\$\hat z_t$};
\node[bxb] (head) at (55.25,41.25) [minimum width=15.5mm, minimum height=7.5mm]
     {Relevance\\head $\Phi$};
\node[bxb] (gate) at (72.75,41.25) [minimum width=15.5mm, minimum height=7.5mm]
     {Budget gate\\$\tau_\beta$};

\draw[arb] (8.6,41.25) -- (enc.west);
\draw[arb] (enc) -- (wm);
\draw[arb] (wm) -- (head);
\draw[arb] (head) -- (gate);

% ------------------------------------------------- routes into the selector
% o_t: the real frame, uploaded only when u_t = 1
\draw[arb] (5,39.05) -- (5,33.0) -- (50,33.0) -- (50,30.5);
\node[lblb, fill=fgBlue!5, inner sep=0.3mm] at (26,33.0) {$o_t$ real frame};

% \hat o_t: the imagined frame, used on every suppressed step
\draw[arb] (37.25,37.5) -- (37.25,35.3) -- (58,35.3) -- (58,30.5);
\node[lblb, fill=fgBlue!5, inner sep=0.3mm] at (47.6,35.3) {$\hat o_t$ predicted};

% u_t: the decision itself
\draw[arb] (72.75,37.5) -- (72.75,30.5);
\node[lblb, anchor=west] at (73.6,34.0) {$u_t$};

% ---------------------------------------------------------------- selector
\node[bxb] (sel) at (62,27.0) [minimum width=36mm, minimum height=7mm]
     {\textcolor{fgMuted}{selector}\\[-0.2mm]
      $\tilde o_t = u_t\,o_t + (1-u_t)\,\hat o_t$};

% -------------------------------------------------------- always-real lane
\node[bxg] (wp) at (20,27.0) [minimum width=36mm, minimum height=7mm]
     {wrist view $+$ proprioception\\[-0.2mm]
      \textcolor{fgGrey}{always real, never scheduled}};

% ------------------------------------------------------------ to the edge
\draw[arb] (62,23.5) -- (62,21.0) -- (30,21.0) -- (30,16.5);
\node[lblb, anchor=west] at (63,21.0) {uplink};
\draw[arg] (20,23.5) -- (20,16.5);

% ------------------------------------------------------------- edge boxes
\node[bxg] (pol) at (22,12.25) [minimum width=32mm, minimum height=8.5mm]
     {Frozen policy $\pi_{0.5}$};
\node[bxg] (chunk) at (63,12.25) [minimum width=34mm, minimum height=8.5mm]
     {Action chunk\\$A_t\in\mathbb{R}^{10\times 7}$};
\draw[arg] (pol) -- (chunk);

% ------------------------------------------------ downlink and action loop
% The executed action advances the world model: suppressing a frame changes
% what the robot will see next, which is why this arrow closes the loop.
\draw[arg] (80,12.25) -- (82.5,12.25) -- (82.5,46.6) -- (37.25,46.6)
           -- (37.25,45.0);
\node[lbl, fill=fgBlue!5, inner sep=0.3mm] at (60,46.6)
     {action downlink: executed $a_{t-1}$};

\end{tikzpicture}
